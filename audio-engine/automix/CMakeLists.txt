if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.25)
  project(TwilightAutoMix LANGUAGES C CXX)
  option(TAE_BUILD_TESTS "Build AutoMix differential tests" ON)
  enable_testing()
endif()
option(TAE_AM_EXPERIMENTAL_PLAYER "Allow in-app AutoMix experimental playback for manual acceptance" OFF)
set(TAE_AM_RECOVERED
  recovered/am_json.c recovered/am_scoring.c recovered/am_analysis.c
  recovered/am_bridge.c recovered/am_timing.c)
# Byte initializers preserve the assets and avoid MSVC's string-literal size limit.
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/assets/TransitionStyles.json TAE_AM_CATALOGUE HEX)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/assets/dsp-parameters.json TAE_AM_PARAMETERS HEX)
string(REGEX REPLACE "(..)" "char(0x\\1),\n" TAE_AM_CATALOGUE "${TAE_AM_CATALOGUE}")
string(REGEX REPLACE "(..)" "char(0x\\1),\n" TAE_AM_PARAMETERS "${TAE_AM_PARAMETERS}")
configure_file(EmbeddedAssets.h.in EmbeddedAssets.h @ONLY)
set(TAE_AM_ASM_AVAILABLE OFF)
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND CMAKE_C_COMPILER_ID STREQUAL "GNU")
  enable_language(ASM)
  set(TAE_AM_ASM_AVAILABLE ON)
endif()
add_library(twilight_automix STATIC ${TAE_AM_RECOVERED} Kernels.c FeatureFrontend.cpp FeaturePostprocess.cpp AutoMix.cpp CandidatePlanner.cpp TransitionEffects.cpp Hash256.cpp ModelRuntime.cpp)
target_compile_features(twilight_automix PRIVATE cxx_std_20)
target_include_directories(twilight_automix PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(twilight_automix PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(twilight_automix PRIVATE ${CMAKE_DL_LIBS})
set_target_properties(twilight_automix PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON POSITION_INDEPENDENT_CODE ON)
if(TAE_AM_ASM_AVAILABLE)
  target_sources(twilight_automix PRIVATE x64/scoring.S x64/selection.S x64/mix.S x64/timing.S)
  target_compile_definitions(twilight_automix PUBLIC TAE_AM_ASM=1)
  if(TAE_AM_EXPERIMENTAL_PLAYER)
    target_compile_definitions(twilight_automix PUBLIC TAE_AM_EXPERIMENTAL_PLAYER=1)
  endif()
endif()
if(MSVC)
  # MSVC requires an explicit opt-in for the C11 atomic function pointer in Kernels.c.
  # Apply this to the source: Visual Studio cannot separate C/C++ target flags.
  set_property(SOURCE Kernels.c APPEND PROPERTY COMPILE_OPTIONS /experimental:c11atomics)
else()
  target_compile_options(twilight_automix PRIVATE $<$<COMPILE_LANGUAGE:C,CXX>:-fno-fast-math;-ffp-contract=off>)
  target_link_libraries(twilight_automix PUBLIC m)
endif()
if(TAE_BUILD_TESTS)
  # Separate symbols keep the untouched arithmetic oracle out of release targets.
  add_library(twilight_automix_oracle STATIC ${TAE_AM_RECOVERED})
  set_target_properties(twilight_automix_oracle PROPERTIES C_STANDARD 11)
  if(MSVC)
    target_compile_options(twilight_automix_oracle PRIVATE /FI${CMAKE_CURRENT_SOURCE_DIR}/recovered/reference_prefix.h)
  else()
    target_compile_options(twilight_automix_oracle PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/recovered/reference_prefix.h -fno-fast-math -ffp-contract=off)
  endif()
  add_executable(twilight_automix_tests tests/differential.cpp)
  target_compile_features(twilight_automix_tests PRIVATE cxx_std_20)
  target_compile_definitions(twilight_automix_tests PRIVATE AM_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
  target_link_libraries(twilight_automix_tests PRIVATE twilight_automix twilight_automix_oracle)
  if(MSVC)
    target_compile_options(twilight_automix_tests PRIVATE /UNDEBUG)
  else()
    target_compile_options(twilight_automix_tests PRIVATE -UNDEBUG -fno-fast-math -ffp-contract=off)
  endif()
  add_test(NAME twilight_automix_differential COMMAND twilight_automix_tests)
  add_executable(twilight_automix_preparation tests/preparation.cpp)
  target_compile_features(twilight_automix_preparation PRIVATE cxx_std_20)
  target_link_libraries(twilight_automix_preparation PRIVATE twilight_automix)
  if(MSVC)
    target_compile_options(twilight_automix_preparation PRIVATE /UNDEBUG)
  else()
    target_compile_options(twilight_automix_preparation PRIVATE -UNDEBUG)
  endif()
  # Keep every preparation assertion, with separate cases so MSVC Debug work
  # fits the existing per-test timeout and failures identify the exact workload.
  add_test(NAME twilight_automix_preparation COMMAND twilight_automix_preparation basic 0)
  foreach(template_index RANGE 0 23)
    add_test(NAME twilight_automix_preparation_template_${template_index}
      COMMAND twilight_automix_preparation template ${template_index})
  endforeach()
  foreach(rate 44100 48000 88200 96000 176400 192000)
    add_test(NAME twilight_automix_preparation_rate_${rate}
      COMMAND twilight_automix_preparation rate ${rate})
  endforeach()
  foreach(channels 1 2)
    add_test(NAME twilight_automix_preparation_budget_${channels}
      COMMAND twilight_automix_preparation budget ${channels})
  endforeach()
  add_executable(twilight_automix_candidates tests/candidates.cpp)
  target_compile_features(twilight_automix_candidates PRIVATE cxx_std_20)
  target_link_libraries(twilight_automix_candidates PRIVATE twilight_automix)
  if(MSVC)
    target_compile_options(twilight_automix_candidates PRIVATE /UNDEBUG)
  else()
    target_compile_options(twilight_automix_candidates PRIVATE -UNDEBUG)
  endif()
  add_test(NAME twilight_automix_candidates COMMAND twilight_automix_candidates)
  add_executable(twilight_automix_plan_probe tests/plan_probe.cpp)
  target_compile_features(twilight_automix_plan_probe PRIVATE cxx_std_20)
  target_link_libraries(twilight_automix_plan_probe PRIVATE twilight_automix)
  add_executable(twilight_automix_frontend_parity tests/frontend_parity.cpp)
  target_compile_features(twilight_automix_frontend_parity PRIVATE cxx_std_20)
  target_link_libraries(twilight_automix_frontend_parity PRIVATE twilight_automix)
  if(MSVC)
    target_compile_options(twilight_automix_frontend_parity PRIVATE /UNDEBUG)
  else()
    target_compile_options(twilight_automix_frontend_parity PRIVATE -UNDEBUG)
  endif()
  # Parity requires original-model fixtures; omission is explicitly untested.
  set(TAE_AM_BEAT_FIXTURES "" CACHE PATH "Beat This conversion fixture directory")
  set(TAE_AM_YAMNET_FIXTURES "" CACHE PATH "YAMNet conversion fixture directory")
  if(TAE_AM_BEAT_FIXTURES)
    add_test(NAME twilight_automix_beat_frontend COMMAND twilight_automix_frontend_parity beat-this ${TAE_AM_BEAT_FIXTURES})
  endif()
  if(TAE_AM_YAMNET_FIXTURES)
    add_test(NAME twilight_automix_yamnet_frontend COMMAND twilight_automix_frontend_parity yamnet ${TAE_AM_YAMNET_FIXTURES})
  endif()
  add_executable(twilight_automix_model_parity tests/model_parity.cpp)
  target_compile_features(twilight_automix_model_parity PRIVATE cxx_std_20)
  target_link_libraries(twilight_automix_model_parity PRIVATE twilight_automix)
  if(MSVC)
    target_compile_options(twilight_automix_model_parity PRIVATE /UNDEBUG)
  else()
    target_compile_options(twilight_automix_model_parity PRIVATE -UNDEBUG)
  endif()
  if(TAE_AM_BEAT_FIXTURES AND TAE_AM_YAMNET_FIXTURES)
    add_test(NAME twilight_automix_model_parity COMMAND twilight_automix_model_parity ${TAE_AM_BEAT_FIXTURES} ${TAE_AM_YAMNET_FIXTURES})
  endif()
endif()
